# OpenEM - open remake of Eishockey Manager (Software 2000, 1993).
# Build (max. 3 parallel compilers on this machine):
#   cmake -S . -B build -G "Visual Studio 18 2026"
#   cmake --build build --config Release -- -m:3 -p:CL_MPCount=1 -v:minimal
#   ctest --test-dir build -C Release -j3
cmake_minimum_required(VERSION 3.24)
project(OpenEM VERSION 0.0.1 LANGUAGES CXX C)
add_compile_definitions(OPENEM_VERSION="${PROJECT_VERSION}")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8 /W4 /permissive-)
endif()

# SDL2: pinned release fetched at configure time (same pin as OpenBMH)

    find_package(SDL2 REQUIRED CONFIG)
    set(EM_SDL_TARGET SDL2::SDL2)
else()
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL2
        GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
        GIT_TAG release-2.32.10
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(SDL2)
    set(EM_SDL_TARGET SDL2::SDL2-static)
endif()

    list(PREPEND EM_SDL_TARGET SDL2::SDL2main)
endif()

    include(FetchContent)
    set(UNICORN_ARCH "x86" CACHE STRING "" FORCE)
    set(UNICORN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(UNICORN_INSTALL OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)

        GIT_REPOSITORY https://github.com/unicorn-engine/unicorn.git
        GIT_TAG 2.1.3
        GIT_SHALLOW TRUE)

    add_test(NAME realcode_market COMMAND em_realcode_market_tests "${EM_REFERENCE}/MONAT1.HOK" "${CMAKE_CURRENT_SOURCE_DIR}/re/out/hok_layout.csv" "${EM_DATA}")
    # whole-program differential test: the original played (in the emulator, through its menu) 9 days from a savegame; the game must reach
    # the same 47 761 state bytes with identical Borland and game random draws (periods of 4, 3, 2 and 2 days: home and away league games, a DEB cup round)
    foreach(period "1;2;4;2" "2;3;3;3" "3;4;2;4" "4;5;2;5" "5;6;3;6")
        list(GET period 0 from)
        list(GET period 1 to)
        list(GET period 2 days)
        list(GET period 3 seed)
        add_test(NAME realcode_daydiff_period${to} COMMAND em_realcode_daydiff "${EM_DATA}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/daydiff/mid${from}.HOK" "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/daydiff/day${to}.bin" ${seed})
        set_tests_properties(realcode_daydiff_period${to} PROPERTIES ENVIRONMENT "GSTATE=${seed};FORCEDAYS=${days}")
    endforeach()
endif()
