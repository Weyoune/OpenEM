# Gemischtes = tile 6 (197,168), Laden = item 0, Speichern = item 1; the file name field at (150,199).

expect screen=NewManager

click 150 60

# --- first half of the season, then save and load as .OEM --------------------------------------------------
play until date>=15.12.1993

# --- on to the play-offs, then save and load as native .HOK -------------------------------------------------
play until date>=01.03.1994
main
click 197 168
click 79 220
click 150 199
type RUN1.HOK
key return
expect screen=Main
main
expect date=21.03.1994
# the original refuses a save of the same day ("Keinen heutigen Spielstand laden !", timed notice)
click 197 168
click 26 220
click 150 199
type D:\RUN1.HOK
key return
expect notice~Keinen heutigen
main
# the guard holds for every save of the running program; after a restart (with the AUTOSAVE.HOK of the game)
# the save loads through the file dialog and the game goes back to its day
restart AUTOSAVE.HOK
main
click 197 168
click 26 220
click 150 199
type D:\RUN1.HOK
key return
expect screen=Main
expect date=21.03.1994
echo HOK geladen
tour

# --- end of the first season: review and its messages, summer, into the second season -----------------------
play until modal=review-overview
render review1-overview.bmp
click 160 120
render review1.bmp

main
expect idle
tour
